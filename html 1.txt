</head>
<body>
    <div class="container">
        <button id="startCallBtn">Start Call</button>
        <button id="endCallBtn">End Call</button>
        <audio id="remoteAudio" controls autoplay></audio>
    </div>

    <script src="https://cdnjs.cloudflare.com/ajax/libs/socket.io/4.0.0/socket.io.js"></script>
    <script>
        const socket = io.connect();

        let localStream;
        let peerConnection;
        const remoteAudio = document.getElementById('remoteAudio');
        const startCallBtn = document.getElementById('startCallBtn');
        const endCallBtn = document.getElementById('endCallBtn');

        const configuration = {
            iceServers: [
                { urls: 'stun:stun.l.google.com:19302' }  // Google's public STUN server
            ]
        };

        startCallBtn.addEventListener('click', async () => {
            // Get the local audio stream from the user's microphone
            localStream = await navigator.mediaDevices.getUserMedia({ audio: true });

            // Initialize the peer connection
            peerConnection = new RTCPeerConnection(configuration);

            // Add the local audio stream to the peer connection
            localStream.getTracks().forEach(track => peerConnection.addTrack(track, localStream));

            // Handle receiving the remote stream
            peerConnection.ontrack = event => {
                const [remoteStream] = event.streams;
                remoteAudio.srcObject = remoteStream;
            };

            // Send ICE candidates to the other peer
            peerConnection.onicecandidate = event => {
                if (event.candidate) {
                    socket.emit('signal', { type: 'candidate', candidate: event.candidate });
                }
            };

            // Create an offer and send it to the other peer
            const offer = await peerConnection.createOffer();
            await peerConnection.setLocalDescription(offer);
            socket.emit('signal', { type: 'offer', offer: peerConnection.localDescription });
        });

        // End the call by closing the peer connection
        endCallBtn.addEventListener('click', () => {
            if (peerConnection) {
                peerConnection.close();
                peerConnection = null;
            }
        });

        // Handle receiving signaling data from the server
        socket.on('signal', async data => {
            if (data.type === 'offer') {
                // If an offer is received, create an answer
                peerConnection = new RTCPeerConnection(configuration);

                // Add the local audio stream to the peer connection
                localStream.getTracks().forEach(track => peerConnection.addTrack(track, localStream));

                // Handle receiving the remote stream
                peerConnection.ontrack = event => {
                    const [remoteStream] = event.streams;
                    remoteAudio.srcObject = remoteStream;
                };

                peerConnection.onicecandidate = event => {
                    if (event.candidate) {
                        socket.emit('signal', { type: 'candidate', candidate: event.candidate });
                    }
                };

                await peerConnection.setRemoteDescription(data.offer);
                const answer = await peerConnection.createAnswer();
                await peerConnection.setLocalDescription(answer);
                socket.emit('signal', { type: 'answer', answer: peerConnection.localDescription });
            } else if (data.type === 'answer') {
                await peerConnection.setRemoteDescription(data.answer);
            } else if (data.type === 'candidate') {
                try {
                    await peerConnection.addIceCandidate(data.candidate);
                } catch (e) {
                    console.error('Error adding received ICE candidate', e);
                }
            }
        });
    </script>
